1.           445. Add Two Numbers II



class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        s1, s2 = [], []
        
        while l1:
            s1.append(l1.val)
            l1 = l1.next
            
        while l2:
            s2.append(l2.val)
            l2 = l2.next
            
        carry = 0
        head = None
        
        while s1 or s2 or carry:
            total = carry
            if s1:
                total += s1.pop()
            if s2:
                total += s2.pop()
                
            carry, val = divmod(total, 10)
            
            node = ListNode(val)
            node.next = head
            head = node
            
        return head



2.          2555. Maximize Win From Two Segments



class Solution:
    def maximizeWin(self, prizePositions: List[int], k: int) -> int:
        n = len(prizePositions)
        dp = [0] * (n + 1)
        max_prizes = 0
        left = 0
        
        for right in range(n):
            while prizePositions[right] - prizePositions[left] > k:
                left += 1
                
            current_segment_count = right - left + 1
            dp[right + 1] = max(dp[right], current_segment_count)
            max_prizes = max(max_prizes, current_segment_count + dp[left])
            
        return max_prizes



3.167. Two Sum II - Input Array Is Sorted



class Solution:
    def twoSum(self, numbers: List[int], target: int) -> List[int]:
        left = 0
        right = len(numbers) - 1
        
        while left < right:
            current_sum = numbers[left] + numbers[right]
            
            if current_sum == target:
                return [left + 1, right + 1]
            elif current_sum < target:
                left += 1
            else:
                right -= 1




4.              1031. Maximum Sum of Two Non-Overlapping Subarrays


class Solution:
    def maxSumTwoNoOverlap(self, nums: List[int], firstLen: int, secondLen: int) -> int:
        def helper(l1: int, l2: int) -> int:
            max_l1 = 0
            ans = 0
            
            s1 = sum(nums[:l1])
            s2 = sum(nums[l1:l1+l2])
            
            max_l1 = s1
            ans = max_l1 + s2
            
            for i in range(len(nums) - l1 - l2):
                s1 += nums[i + l1] - nums[i]
                s2 += nums[i + l1 + l2] - nums[i + l1]
                max_l1 = max(max_l1, s1)
                ans = max(ans, max_l1 + s2)
                
            return ans
            
        return max(helper(firstLen, secondLen), helper(secondLen, firstLen))




5.      1305. All Elements in Two Binary Search Trees



class Solution:
    def getAllElements(self, root1: TreeNode, root2: TreeNode) -> List[int]:
        def inorder(root: TreeNode, res: List[int]):
            stack = []
            curr = root
            while curr or stack:
                while curr:
                    stack.append(curr)
                    curr = curr.left
                curr = stack.pop()
                res.append(curr.val)
                curr = curr.right

        list1, list2 = [], []
        inorder(root1, list1)
        inorder(root2, list2)
        
        merged = []
        i, j = 0, 0
        while i < len(list1) and j < len(list2):
            if list1[i] < list2[j]:
                merged.append(list1[i])
                i += 1
            else:
                merged.append(list2[j])
                j += 1
                
        merged.extend(list1[i:])
        merged.extend(list2[j:])
        return merged




6.      2131. Longest Palindrome by Concatenating Two Letter Words



from collections import Counter

class Solution:
    def longestPalindrome(self, words: list[str]) -> int:
        counts = Counter(words)
        length = 0
        has_central_pivot = False
        
        for word, count in counts.items():
            if word[0] == word[1]:
                if count % 2 == 0:
                    length += count * 2
                else:
                    length += (count - 1) * 2
                    has_central_pivot = True
            else:
                reversed_word = word[::-1]
                if reversed_word in counts:
                    pairs = min(count, counts[reversed_word])
                    length += pairs * 2
                    
        if has_central_pivot:
            length += 2
            
        return length




7.          3361. Shift Distance Between Two Strings


class Solution:
    def shiftDistance(self, s: str, t: str, nextCost: List[int], previousCost: List[int]) -> int:
        forward_pref = [0] * 53
        backward_pref = [0] * 53
        
        for i in range(52):
            forward_pref[i + 1] = forward_pref[i] + nextCost[i % 26]
            backward_pref[i + 1] = backward_pref[i] + previousCost[i % 26]
            
        total_cost = 0
        
        for char_s, char_t in zip(s, t):
            u = ord(char_s) - 97
            v = ord(char_t) - 97
            
            if u == v:
                continue
                
            if v > u:
                f_cost = forward_pref[v] - forward_pref[u]
                b_cost = backward_pref[u + 26] - backward_pref[v]
            else:
                f_cost = forward_pref[v + 26] - forward_pref[u]
                b_cost = backward_pref[u] - backward_pref[v]
                
            total_cost += min(f_cost, b_cost)
            
        return total_cost



8.          1616. Split Two Strings to Make Palindrome


class Solution:
    def checkPalindromeFormation(self, a: str, b: str) -> bool:
        def is_palindrome(s: str, i: int, j: int) -> bool:
            while i < j:
                if s[i] != s[j]:
                    return False
                i += 1
                j -= 1
            return True

        def check(s1: str, s2: str) -> bool:
            i, j = 0, len(s1) - 1
            while i < j and s1[i] == s2[j]:
                i += 1
                j -= 1
            return is_palindrome(s1, i, j) or is_palindrome(s2, i, j)

        return check(a, b) or check(b, a)



9.          1657. Determine if Two Strings Are Close




from collections import Counter

class Solution:
    def closeStrings(self, word1: str, word2: str) -> bool:
        if len(word1) != len(word2):
            return False
            
        c1 = Counter(word1)
        c2 = Counter(word2)
        
        if set(c1.keys()) != set(c2.keys()):
            return False
            
        return sorted(c1.values()) == sorted(c2.values())




10.             2513. Minimize the Maximum of Two Arrays
    


import math

class Solution:
    def minimizeSet(self, divisor1: int, divisor2: int, uniqueCnt1: int, uniqueCnt2: int) -> int:
        lcm = (divisor1 * divisor2) // math.gcd(divisor1, divisor2)
        
        low = 1
        high = 10**10
        ans = high
        
        while low <= high:
            mid = (low + high) // 2
            
            not_div1 = mid - (mid // divisor1)
            not_div2 = mid - (mid // divisor2)
            not_div_both = mid - (mid // lcm)
            
            if not_div1 >= uniqueCnt1 and not_div2 >= uniqueCnt2 and not_div_both >= (uniqueCnt1 + uniqueCnt2):
                ans = mid
                high = mid - 1
            else:
                low = mid + 1
                
        return ans
