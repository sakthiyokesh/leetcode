1.1998. GCD Sort of an Array

from collections import defaultdict

class Solution:
    def gcdSort(self, nums: list[int]) -> bool:
        max_num = max(nums)
        parent = list(range(max_num + 1))
        
        def find(i: int) -> int:
            if parent[i] == i:
                return i
            parent[i] = find(parent[i])
            return parent[i]
            
        def union(i: int, j: int) -> None:
            root_i = find(i)
            root_j = find(j)
            if root_i != root_j:
                parent[root_i] = root_j

        visited = [False] * (max_num + 1)
        for num in nums:
            visited[num] = True

        for i in range(2, max_num + 1):
            first_val = -1
            for j in range(i, max_num + 1, i):
                if visited[j]:
                    if first_val == -1:
                        first_val = j
                    else:
                        union(first_val, j)
                        
        sorted_nums = sorted(nums)
        for original, target in zip(nums, sorted_nums):
            if original != target and find(original) != find(target):
                return False
                
        return True


2.  761. Special Binary String

class Solution:
    def makeLargestSpecial(self, s: str) -> str:
        count = 0
        i = 0
        res = []
        
        for j in range(len(s)):
            if s[j] == '1':
                count += 1
            else:
                count -= 1
                
            if count == 0:
                res.append("1" + self.makeLargestSpecial(s[i + 1:j]) + "0")
                i = j + 1
                
        res.sort(reverse=True)
        return "".join(res)


3. 1363. Largest Multiple of Three


class Solution:
    def largestMultipleOfThree(self, digits: list[int]) -> str:
        count = [0] * 10
        total_sum = 0
        for d in digits:
            count[d] += 1
            total_sum += d
            
        remainder = total_sum % 3
        
        if remainder != 0:
            groups = [[] for _ in range(3)]
            for d in range(10):
                for _ in range(count[d]):
                    groups[d % 3].append(d)
                    
            removed = False
            if remainder == 1:
                if len(groups[1]) >= 1:
                    count[groups[1][0]] -= 1
                    removed = True
                elif len(groups[2]) >= 2:
                    count[groups[2][0]] -= 1
                    count[groups[2][1]] -= 1
                    removed = True
            elif remainder == 2:
                if len(groups[2]) >= 1:
                    count[groups[2][0]] -= 1
                    removed = True
                elif len(groups[1]) >= 2:
                    count[groups[1][0]] -= 1
                    count[groups[1][1]] -= 1
                    removed = True
                    
            if not removed:
                return ""
                
        result = []
        for d in range(9, -1, -1):
            result.append(str(d) * count[d])
            
        ans = "".join(result)
        
        if ans and ans[0] == "0":
            return "0"
            
        return ans


4.  1520. Maximum Number of Non-Overlapping Substrings


class Solution:
    def maxNumOfSubstrings(self, s: str) -> list[str]:
     
        first = {}
        last = {}
        for i, ch in enumerate(s):
            if ch not in first:
                first[ch] = i
            last[ch] = i
            
        valid_intervals = []
        
    
        for ch in first:
            start = first[ch]
            end = last[ch]
            
            is_valid = True
            curr = start
            while curr <= end:
                char_at_curr = s[curr]
                
                if first[char_at_curr] < start:
                    is_valid = False
                    break
                end = max(end, last[char_at_curr])
                curr += 1
                
            if is_valid:
                valid_intervals.append((start, end))
                
       
        valid_intervals.sort(key=lambda x: x[1])
        
        ans = []
        prev_end = -1
        
     
        for start, end in valid_intervals:
           
            if start > prev_end:
                ans.append(s[start:end + 1])
                prev_end = end
           
            elif start >= first[s[prev_end]] and end <= prev_end:
                ans[-1] = s[start:end + 1]
                prev_end = end
                
        return ans


5.  32. Longest Valid Parentheses

class Solution:
    def longestValidParentheses(self, s: str) -> int:
        stack = [-1]
        max_length = 0
        
        for i, char in enumerate(s):
            if char == '(':
                stack.append(i)
            else:
                stack.pop()
                if not stack:
                    stack.append(i)
                else:
                    max_length = max(max_length, i - stack[-1])
                    
        return max_length


6. 52. N-Queens II


class Solution:
    def totalNQueens(self, n: int) -> int:
        cols = set()
        pos_diag = set()
        neg_diag = set()
        
        def backtrack(r: int) -> int:
            if r == n:
                return 1
                
            solutions = 0
            for c in range(n):
                if c in cols or (r + c) in pos_diag or (r - c) in neg_diag:
                    continue
                    
                cols.add(c)
                pos_diag.add(r + c)
                neg_diag.add(r - c)
                
                solutions += backtrack(r + 1)
                
                cols.remove(c)
                pos_diag.remove(r + c)
                neg_diag.remove(r - c)
                
            return solutions
            
        return backtrack(0)


7. 60. Permutation Sequence

import math

class Solution:
    def getPermutation(self, n: int, k: int) -> str:
        numbers = [str(i) for i in range(1, n + 1)]
        k -= 1
        
        result = []
        for i in range(n - 1, -1, -1):
            fact = math.factorial(i)
            idx = k // fact
            result.append(numbers.pop(idx))
            k %= fact
            
        return "".join(result)


8.  65. Valid Number

class Solution:
    def isNumber(self, s: str) -> bool:
        seen_digit = False
        seen_dot = False
        seen_exponent = False
        
        for i, char in enumerate(s):
            if char.isdigit():
                seen_digit = True
            elif char in ('+', '-'):
                if i > 0 and s[i - 1] not in ('e', 'E'):
                    return False
            elif char in ('e', 'E'):
                if seen_exponent or not seen_digit:
                    return False
                seen_exponent = True
                seen_digit = False
            elif char == '.':
                if seen_dot or seen_exponent:
                    return False
                seen_dot = True
            else:
                return False
                
        return seen_digit


9.  2172. Maximum AND Sum of Array


from functools import lru_cache

class Solution:
    def maximumANDSum(self, nums: list[int], numSlots: int) -> int:
        n = len(nums)
        
        @lru_cache(None)
        def dfs(i: int, mask: int) -> int:
          
            if i == n:
                return 0
            
            max_sum = 0
           
            for slot in range(1, numSlots + 1):
                
                bit1 = (slot - 1) * 2
                bit2 = bit1 + 1
                
              
                if not (mask & (1 << bit1)):
                    current_and = nums[i] & slot
                    max_sum = max(max_sum, current_and + dfs(i + 1, mask | (1 << bit1)))
              
                elif not (mask & (1 << bit2)):
                    current_and = nums[i] & slot
                    max_sum = max(max_sum, current_and + dfs(i + 1, mask | (1 << bit2)))
                    
            return max_sum

       
        return dfs(0, 0)

10.  2366. Minimum Replacements to Sort the Array


class Solution:
    def minimumReplacement(self, nums: list[int]) -> int:
        operations = 0
        n = len(nums)
        
       
        next_max_allowed = nums[-1]
        
        
        for i in range(n - 2, -1, -1):
            curr = nums[i]
            
           
            if curr <= next_max_allowed:
                next_max_allowed = curr
                continue
            
            
            num_pieces = (curr + next_max_allowed - 1) // next_max_allowed
            
           
            operations += num_pieces - 1
            
            
            next_max_allowed = curr // num_pieces
            
        return operations
